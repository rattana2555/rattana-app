/* ========== รับคืนสินค้า (CN sheet) — โครงตามแอป Glide เก่า ========== */
const RETURN_TYPES=['คืนเต็มทั้งบิล','คืนบ้างบิล'];
const BILL_TYPES=['DC (ปลีก)','DS (ส่ง)'];
// สาเหตุ 2 ชั้น (category → sub)
const REASON_TREE={
  'รถส่ง':['สินค้าชำรุด(จากขนส่ง)','สินค้าตกหล่น','ส่งผิดออเดอร์'],
  'ร้านค้า':['ปิดร้าน','ไม่รับสินค้า','สั่งผิด','ร้านค้าคืน'],
  'คลัง':['สินค้าหมดอายุ','สินค้าชำรุด(จากคลัง)','จ่ายผิดสินค้า','จ่ายผิดจำนวน'],
  'เซลล์':['ขายผิด','ราคาผิด','โปรโมชั่นผิด'],
  'อื่นๆ':['อื่นๆ'],
};
const ITEM_UNITS=['ชิ้น(EA)','แพ็ค(CS)'];
const returnState={
  photo:'',
  returnType:'', billType:'',
  customerCode:'', customerName:'',
  docNo:'',
  loaderName:'',
  driverName:'',
  reasonCategory:'', reasonSub:'',
  note:'',
  items:[{productId:'',productName:'',unit:'ชิ้น(EA)',qty:'',price:''}],
};

function returnGrandTotal(){
  return returnState.items.reduce((s,it)=>s+(parseFloat(it.qty)||0)*(parseFloat(it.price)||0),0);
}

function renderReturn(){
  const drivers=state.drivers.map(d=>{
    const nk=cleanCell(d['ชื่อเล่น']); const fn=cleanCell(d['ชื่อ - สกุล']||d['ชื่อ-สกุล']||'');
    const full=fn?`${fn} (${nk})`:nk; return {v:full,l:full};
  });
  const customers=state.customers.filter(c=>c.code&&c.name).map(c=>({v:c.code,l:`${c.name} — ${c.code}`}));
  const subReasons=returnState.reasonCategory?(REASON_TREE[returnState.reasonCategory]||[]):[];

  $('returnFormBody').innerHTML=`
    <div class="field">
      <label>รูปสินค้าคืน <span class="txt-xs txt-muted">(ใส่ภาพการคืนสินค้าที่นี่)</span></label>
      <div class="photo-controls">
        <label class="btn btn-outline btn-sm" style="cursor:pointer">📷 กล้อง
          <input type="file" accept="image/*" capture="environment" hidden onchange="handleReturnPhoto(event)">
        </label>
        <label class="btn btn-outline btn-sm" style="cursor:pointer">🖼 อัลบั้ม
          <input type="file" accept="image/*" hidden onchange="handleReturnPhoto(event)">
        </label>
      </div>
      <div class="photo-preview" id="returnPhotoPreview">${returnState.photo?renderReturnThumb():''}</div>
    </div>

    <div class="field">
      <label>ประเภทการคืน</label>
      <div class="pills" id="returnTypePills">
        ${RETURN_TYPES.map(t=>`<button type="button" class="pill ${t===returnState.returnType?'active gold':''}" data-v="${esc(t)}" onclick="pickReturnType(this)">${esc(t)}</button>`).join('')}
      </div>
    </div>

    <div class="field">
      <label>ประเภทบิล</label>
      <div class="pills" id="billTypePills">
        ${BILL_TYPES.map(t=>`<button type="button" class="pill ${t===returnState.billType?'active gold':''}" data-v="${esc(t)}" onclick="pickBillType(this)">${esc(t)}</button>`).join('')}
      </div>
    </div>

    <div class="field">
      <label>ชื่อร้านค้า</label>
      ${buildDD('ddReturnCust',customers,'-- ค้นหาชื่อร้าน --')}
    </div>

    <div class="field">
      <label>เลขที่เอกสาร</label>
      <input class="input" id="returnDocNo" value="${esc(returnState.docNo)}" oninput="returnState.docNo=this.value" placeholder="DSI..../00xxx">
    </div>

    <div class="field">
      <label>ชื่อพนักงานขนของ</label>
      ${buildDD('ddReturnLoader',drivers,'-- เลือกพนักงาน --')}
    </div>

    <div class="field">
      <label>ชื่อผู้ขับ</label>
      ${buildDD('ddReturnDriver',drivers,'-- เลือกผู้ขับ --')}
    </div>

    <div class="field">
      <label>สาเหตุการคืน</label>
      ${buildDD('ddReturnReasonCat',Object.keys(REASON_TREE).map(k=>({v:k,l:k})),'-- เลือกสาเหตุ --')}
    </div>

    <div class="field" id="reasonSubWrap" style="${returnState.reasonCategory?'':'display:none'}">
      <label>รายละเอียดสาเหตุ</label>
      ${buildDD('ddReturnReasonSub',subReasons.map(s=>({v:s,l:s})),'-- เลือก --')}
    </div>

    <div class="field">
      <label>หมายเหตุ</label>
      <textarea class="input" id="returnNote" placeholder="ระบุเพิ่มเติม (ถ้ามี)" oninput="returnState.note=this.value">${esc(returnState.note)}</textarea>
    </div>

    <div class="divider"></div>
    <h3 class="mb-8">📋 รายการสินค้าคืน <span class="txt-sm txt-muted">(สูงสุด 5 รายการ)</span></h3>
    <div id="returnItems"></div>
    <button class="btn btn-outline btn-sm mt-8" id="btnAddItem" onclick="addReturnItem()">+ เพิ่มรายการ</button>

    <div class="divider"></div>
    <div class="bills-total" id="returnGrandTotal">
      <span>รวมทั้งหมด</span>
      <span class="num">0.00 ฿</span>
    </div>
  `;
  if(returnState.customerCode){
    const cust=state.customers.find(c=>c.code===returnState.customerCode);
    if(cust) ddPick('ddReturnCust',cust.code,`${cust.name} — ${cust.code}`);
  }
  if(returnState.loaderName) ddPick('ddReturnLoader',returnState.loaderName,returnState.loaderName);
  if(returnState.driverName) ddPick('ddReturnDriver',returnState.driverName,returnState.driverName);
  if(!returnState.driverName&&state.user.displayName){
    returnState.driverName=state.user.displayName;
    ddPick('ddReturnDriver',state.user.displayName,state.user.displayName);
  }
  if(returnState.reasonCategory) ddPick('ddReturnReasonCat',returnState.reasonCategory,returnState.reasonCategory);
  if(returnState.reasonSub) ddPick('ddReturnReasonSub',returnState.reasonSub,returnState.reasonSub);
  document.querySelectorAll('#ddReturnCust_list li').forEach(li=>{
    li.addEventListener('click',()=>{
      returnState.customerCode=li.dataset.v;

      returnState.customerName=cust?cust.name:'';
    });
  });
  document.querySelectorAll('#ddReturnLoader_list li').forEach(li=>{
    li.addEventListener('click',()=>{ returnState.loaderName=li.dataset.v; });
  });
  document.querySelectorAll('#ddReturnDriver_list li').forEach(li=>{
    li.addEventListener('click',()=>{ returnState.driverName=li.dataset.v; });
  });
  document.querySelectorAll('#ddReturnReasonCat_list li').forEach(li=>{
    li.addEventListener('click',()=>{
      returnState.reasonCategory=li.dataset.v; returnState.reasonSub='';
      renderReturn();
    });
  });

    li.addEventListener('click',()=>{ returnState.reasonSub=li.dataset.v; });
  });
  renderReturnItems();
}

function renderReturnItems(){
  const el=$('returnItems'); if(!el) return;
  el.innerHTML=returnState.items.map((it,i)=>{
    const lineTotal=(parseFloat(it.qty)||0)*(parseFloat(it.price)||0);
    return `<div class="card" style="margin-bottom:8px">
      <div class="card-head" style="background:var(--row-alt)">
        <span class="txt-bold">รายการที่ ${i+1}</span>
        ${returnState.items.length>1?`<button class="btn btn-outline btn-sm txt-danger" onclick="removeReturnItem(${i})">× ลบ</button>`:''}
      </div>
      <div class="card-body">
        <div class="field" style="margin-bottom:8px">
          <label>สินค้า</label>
          <input class="input" value="${esc(it.productName)}" oninput="returnState.items[${i}].productName=this.value" placeholder="ชื่อสินค้า / บาร์โค้ด">
        </div>
        <div class="field" style="margin-bottom:8px">
          <label>หน่วย</label>
          <div class="pills">
            ${ITEM_UNITS.map(u=>`<button type="button" class="pill ${u===it.unit?'active gold':''}" onclick="pickItemUnit(${i},'${esc(u)}')">${esc(u)}</button>`).join('')}
          </div>
        </div>
        <div class="grid-2">
          <div class="field" style="margin-bottom:0">
            <label>จำนวน</label>
            <input class="input input-num" type="number" inputmode="numeric" min="0" value="${esc(it.qty)}" oninput="returnState.items[${i}].qty=this.value;recalcReturnTotal()" placeholder="0">
          </div>
          <div class="field" style="margin-bottom:0">
            <label>ราคา/หน่วย</label>
            <input class="input input-num" type="number" inputmode="decimal" step="any" min="0" value="${esc(it.price)}" oninput="returnState.items[${i}].price=this.value;recalcReturnTotal()" placeholder="0.00">
          </div>
        </div>
        <div class="txt-right txt-sm txt-muted mt-8">รวม <span class="num txt-gold txt-bold">${fmt(lineTotal)}</span></div>
      </div>
    </div>`;
  }).join('');
  $('btnAddItem').disabled=returnState.items.length>=5;
  recalcReturnTotal();
}
function recalcReturnTotal(){
  const t=returnGrandTotal();
  const el=$('returnGrandTotal'); if(el) el.children[1].textContent=`${fmt(t)} ฿`;
}
function pickItemUnit(idx,unit){
  returnState.items[idx].unit=unit;
  renderReturnItems();
}
function addReturnItem(){
  if(returnState.items.length>=5){ showToast('ใส่ได้สูงสุด 5 รายการ','warning'); return; }
  returnState.items.push({productId:'',productName:'',unit:'ชิ้น(EA)',qty:'',price:''});
  renderReturnItems();
}
function removeReturnItem(i){ returnState.items.splice(i,1); renderReturnItems(); }
function pickReturnType(btn){ btn.parentElement.querySelectorAll('.pill').forEach(p=>{p.classList.remove('active','gold');}); btn.classList.add('active','gold'); returnState.returnType=btn.dataset.v; }
function pickBillType(btn){ btn.parentElement.querySelectorAll('.pill').forEach(p=>{p.classList.remove('active','gold');}); btn.classList.add('active','gold'); returnState.billType=btn.dataset.v; }
function renderReturnThumb(){ return `<div class="photo-thumb"><img src="${returnState.photo}"><button class="photo-del" onclick="returnState.photo='';$('returnPhotoPreview').innerHTML=''">×</button></div>`; }
function handleReturnPhoto(ev){
  const f=ev.target.files[0]; if(!f) return;
  const reader=new FileReader();
  reader.onload=e=>{
    const img=new Image();
    img.onload=()=>{
      const max=1280; let w=img.width,h=img.height;
      if(w>max||h>max){ const r=Math.min(max/w,max/h); w*=r; h*=r; }
      const c=document.createElement('canvas'); c.width=w; c.height=h;
      c.getContext('2d').drawImage(img,0,0,w,h);
      returnState.photo=c.toDataURL('image/jpeg',0.78);
      $('returnPhotoPreview').innerHTML=renderReturnThumb();
    };
    img.src=e.target.result;
  };
  reader.readAsDataURL(f);
}
function resetReturnForm(){
  Object.assign(returnState,{
    photo:'',returnType:'',billType:'',
    customerCode:'',customerName:'',docNo:'',
    loaderName:'',driverName:'',
    reasonCategory:'',reasonSub:'',note:'',
    items:[{productId:'',productName:'',unit:'ชิ้น(EA)',qty:'',price:''}]
  });
  renderReturn();
  showToast('ล้างข้อมูลแล้ว','info');
}
async function confirmReturn(){
  if(!returnState.customerCode){ showToast('กรุณาเลือกชื่อร้านค้า','warning'); return; }
  if(!returnState.docNo){ showToast('กรุณาใส่เลขที่เอกสาร','warning'); return; }
  if(!returnState.returnType){ showToast('กรุณาเลือกประเภทการคืน','warning'); return; }
  if(!returnState.reasonCategory){ showToast('กรุณาเลือกสาเหตุการคืน','warning'); return; }
  const job=state.jobs.find(j=>j.bills.some(b=>b.doc===returnState.docNo));
  const total=returnGrandTotal();
  const payload={
    action:'save_return',
    jobId:job?.id||'',
    docNo:returnState.docNo,
    customerCode:returnState.customerCode,
    customerName:returnState.customerName,
    warehouse:job?.warehouse||state.user.warehouse||'',
    returnType:returnState.returnType,
    billType:returnState.billType,
    reason:returnState.reasonSub||returnState.reasonCategory,
    errorFrom:returnState.reasonCategory,
    items:returnState.items.filter(it=>it.productName).slice(0,5).map(it=>({
      barcode:'',name:it.productName,qty:parseFloat(it.qty)||0,
      price:parseFloat(it.price)||0,unit:it.unit
    })),
    totalAmount:total,
    note:returnState.note,
    photo:returnState.photo,
    driver:returnState.driverName||state.user.displayName||state.user.nickname,
    loader:returnState.loaderName||'',
    submittedAt:Date.now(),
  };
  const url=getAppsUrl();
  if(url){
    try{ await fetch(url,{method:'POST',mode:'no-cors',headers:{'Content-Type':'text/plain'},body:JSON.stringify(payload)}); }
    catch(e){ console.warn('return sync failed',e); }
  }
  showToast(`บันทึกรับคืน ${returnState.docNo} ✓`,'success');
  resetReturnForm();
  switchTab('sec-home');
}
